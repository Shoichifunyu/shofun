```python
class Solution:
    def ladderLength(self, beginWord: str, endWord: str, wordList: List[str]) -> int:
        if endWord not in wordList:
            return 0
        
        # beginWordも必要なのでaddする
        # ただし、beginWordも要素に入れた場合、一意性が保証できなくなるので、set型にaddする形とする
        wordSet: Set[str] = set(wordList)
        wordSet.add(beginWord)
        
        # patternごとのwordListを作る
        pattern_to_word = defaultdict(list)
        for word in wordSet:
            for i in range(len(word)):
                pattern = word[:i] + "*" + word[i+1:]
                pattern_to_word[pattern].append(word)
        
        word_to_steps = defaultdict(int)
        word_to_steps[beginWord] = 1

        queue = deque()
        queue.append(word_to_steps)

        appendSet = set()

        while queue:
            word_to_steps = queue.popleft()
            # 途中でword_to_stepsの要素数が変化するとERRORになるので、list化
            for word, steps in word_to_steps.items():
                next_step = steps
                # 同じwordの探索は二度実施しないため、dictから削除する
                # del word_to_steps[word]
                word_to_next_step = {}
                for i in range(len(word)):
                    pattern = word[:i] + "*" + word[i+1:]
                    # 当該patternに対応するwordListを抽出する
                    if word_list := pattern_to_word.get(pattern):
                        # wordListが存在する場合、リストのループで各wordについて深堀する(=次のword候補)
                        for word_candidate in word_list:
                            # wordがendWordに等しかったら処理は終わる
                            if word_candidate == endWord:
                                # endWordにたどり着くための+1
                                next_step = steps + 1
                                return next_step
                            # endWordではなかったら、探索を継続する必要がある為、次のstepとして、当該wordを登録しておく
                            # ただし、当然、word_candidate != wordとなることを期待している
                            if word_candidate not in appendSet:
                                appendSet.add(word_candidate)
                                next_step = steps + 1
                                word_to_next_step[word_candidate] = next_step
                                queue.append(word_to_next_step)
        return 0
```
