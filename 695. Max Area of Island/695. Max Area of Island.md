```python
class Solution:
    def maxAreaOfIsland(self, grid: List[List[int]]) -> int:
        height = len(grid)
        width = len(grid[0])
        def is_water(row: int, col: int) -> bool:
            return (
                not (0 <= row < height and 0 <= col < width)
                    or
                grid[row][col] == 0
            )

        visited = [[False] * width for _ in range(height)]
        walk_count = 0
        def visit_connect_islands(row: int, col: int) -> int:
            visited[row][col] = True
            next_island = [(row, col)]
            offsets = [(-1, 0), (0, -1), (1, 0), (0, 1)]
            if is_water(row, col):
                walk_count = 0
            else:
                walk_count = 1
            while next_island:
                row, col = next_island.pop()
                for row_offset, col_offset in offsets:
                    next_row = row + row_offset
                    next_col = col + col_offset
                    if is_water(next_row, next_col) or visited[next_row][next_col]:
                        continue
                    else:
                        walk_count += 1
                    next_island.append((next_row, next_col))
                    visited[next_row][next_col] = True
            return walk_count
            

        walk_counts = [0]      
        for row, col in product(range(height), range(width)):
            if is_water(row, col) or visited[row][col]:
                continue                
            walk_count = visit_connect_islands(row, col)
            walk_counts.append(walk_count)
            print(walk_counts)
        
        return max(walk_counts)
```
- `walk_counts = [0]`と定義することで、配列の全要素が0の場合にもカバーしている

```python
class Solution:
    def maxAreaOfIsland(self, grid: list[list[int]]) -> int:
        height = len(grid)
        width = len(grid[0])
        is_explored = [[False] * width for _ in range(height)]

        def is_frontier(row: int, col: int) -> bool:
            return (
                (0 <= row < height and 0 <= col < width)
                and grid[row][col] == 1
                and not is_explored[row][col]
            )

        def measure_island(row: int, col: int) -> int:
            if not is_frontier(row, col):
                return 0

            is_explored[row][col] = True
            area = 1
            offsets = [(1, 0), (-1, 0), (0, 1), (0, -1)]
            for row_offset, col_offset in offsets:
                area += measure_island(row + row_offset, col + col_offset)
            return area

        max_area = 0
        for row, col in product(range(height), range(width)):
            max_area = max(max_area, measure_island(row, col))
        return max_area
```

- 再帰を使ったBFSで解く方法があったので、これを使って解くことにした。

```python
class Solution:
    def maxAreaOfIsland(self, grid: List[List[int]]) -> int:
        LAND_CELL = 1
        height = len(grid)
        width = len(grid[0])
        visited = [[False] * width for _ in range(height)]

        def is_frontier(row: int, col: int) -> bool:
            return (
                0 <= row < height and 0 <= col < width
                and grid[row][col] == LAND_CELL
                and not visited[row][col]
            )

        def visit_grid(row: int, col: int, num_of_islands: int) -> int:
            visited[row][col] = True
            offsets = [(-1, 0), (0, -1), (1, 0), (0 ,1)]
            num_of_islands = 1
            for offset_row, offset_col in offsets:
                next_row = row + offset_row
                next_col = col + offset_col
                if is_frontier(next_row, next_col):
                    num_of_islands += visit_grid(next_row, next_col, num_of_islands)
            return num_of_islands

        max_num_of_islands = 0
        for row, col in product(range(height), range(width)):
            if is_frontier(row, col):
                max_num_of_islands = max(max_num_of_islands, visit_grid(row, col, max_num_of_islands))
        return max_num_of_islands
```
- 変数名など、自分流に書き換えたりしたが、基本形はhttps://github.com/nanae772/leetcode-arai60/pull/19 のものと同じ
