```py
# Definition for a binary tree node.
# class TreeNode:
#     def __init__(self, val=0, left=None, right=None):
#         self.val = val
#         self.left = left
#         self.right = right
class Solution:
    def levelOrder(self, root: Optional[TreeNode]) -> List[List[int]]:
        # [[親], [一つ下の左ノード, 一つ下の右ノード], [一つ下の左ノードの下の左ノード, ...]
        # 上記順でレベル順にList[List[int]]を返す

        def assign_depth_to_box(node: TreeNode, depth_to_box: dict[int, list], depth: int):
            if node.left is not None:
                depth_to_box[depth].append(node.left.val)
                assign_depth_to_box(node.left, depth_to_box, depth + 1)
            if node.right is not None:
                depth_to_box[depth].append(node.right.val)
                assign_depth_to_box(node.right, depth_to_box, depth + 1)
            return depth_to_box

        depth_to_box = defaultdict(list)
        if root is None:
            return depth_to_box[0]

        depth_to_box[0].append(root.val)
        depth_to_box = assign_depth_to_box(root, depth_to_box, 1)
        return [inner_box for inner_box in depth_to_box.values() if len(inner_box) > 0]
```
- defaultdict(list)によって、depth(深さ)をkey、listをvalueとする辞書を作る
- その辞書のvalueをリスト内包表記で最後に返す

- 深さを意識したリストへの追加は同じ。一方で、よりシンプルな実装になる方法が存在した。
  - levelが、内部リスト(List[List[int]]のList[int]の部分)の要素数と対応していることに着目した実装
    - rootの要素数は1つ、2番目の層の要素数はその層のList[int]が追加され2つ、3番目までの層の要素数は1, 2, 3番目の要素のList[int]で3つ、4番目は... と続いていく
```py
# Definition for a binary tree node.
# class TreeNode:
#     def __init__(self, val=0, left=None, right=None):
#         self.val = val
#         self.left = left
#         self.right = right
class Solution:
    def levelOrder(self, root: Optional[TreeNode]) -> List[List[int]]:
        def search_each_position_elements(node: TreeNode, ret: list, pos: int) ->  List[List[int]]:
            if pos >= len(ret):
                ret.append([])
            ret[pos].append(node.val)

            if node.left is not None:
                search_each_position_elements(node.left, ret, pos + 1)
            if node.right is not None:
                search_each_position_elements(node.right, ret, pos + 1)
            return ret

        ret = []
        if root is None:
            return ret
        return search_each_position_elements(root, ret, 0)
```

```py
class Solution:
    def levelOrder(self, root: Optional[TreeNode]) -> List[List[int]]:
        ret = []
        if root is None:
            return ret
        def build_each_level_elements(node: TreeNode, pos: int):
            while pos >= len(ret):
                ret.append([])
            ret[pos].append(node.val)

            if node.left is not None:
                build_each_level_elements(node.left, pos + 1)
            if node.right is not None:
                build_each_level_elements(node.right, pos + 1)

        build_each_level_elements(root, 0)
        return ret
```
