# Definition for a binary tree node.
# class TreeNode:
#     def __init__(self, val=0, left=None, right=None):
#         self.val = val
#         self.left = left
#         self.right = right
class Solution:
    def hasPathSum(self, root: Optional[TreeNode], targetSum: int) -> bool:
        def is_child(node: TreeNode) -> bool:
            if node.left is None and node.right is None:
                return True
            return False

        def validate_num(node: TreeNode, targetSum: int) -> boot:
            if is_child(node):
                return targetSum == node.val

            targetSum -= node.val
            bool_left = False
            bool_right = False
            if node.left is not None:
                bool_left = validate_num(node.left, targetSum)
            if node.right is not None:
                bool_right = validate_num(node.right, targetSum)
            return bool_left or bool_right

        if root is None:
            return False
        return validate_num(root, targetSum)

```python
# Definition for a binary tree node.
# class TreeNode:
#     def __init__(self, val=0, left=None, right=None):
#         self.val = val
#         self.left = left
#         self.right = right
class Solution:
    def hasPathSum(self, root: Optional[TreeNode], targetSum: int) -> bool:
        def check_sum(node: TreeNode, target_sum: int) -> bool:
            target_sum -= node.val
            if node.left is None and node.right is None:
                if target_sum == 0:
                    return True
            if node.left is not None:
                if check_sum(node.left, target_sum):
                    return True
            if node.right is not None:
                if check_sum(node.right, target_sum):
                    return True
        if root is None:
            return False
        if check_sum(root, targetSum):
            return True
        return False
```
